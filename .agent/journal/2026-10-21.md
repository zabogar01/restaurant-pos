## The watcher (the job's temp folder is deleted with the session, so it is kept here)

Prompts and approvals do not wake the lead, so a background zsh script did: it exits with one line when an agent's
Handoff ends `DONE`/`BLOCKED` (only after the worktree's HEAD has moved past `BASE_HEAD`, so an earlier round's `DONE`
does not trip it), when the agent has been `blocked` (an approval) for `BLOCK_SEC` seconds (900 kept the lead from
reporting every check-run approval), or when its pane text has not changed for 20 minutes. Run it with
`run_in_background` and a two-hour timeout; macOS `/bin/bash` is 3.2 (no associative arrays) and `status` is read-only
in zsh, hence the names.

    #!/bin/zsh
    # Args: one or more "name|pane|taskfile". Env: BASE_HEAD (optional), BLOCK_SEC (default 10).
    typeset -A last_text last_change blocked_since
    now() { date +%s; }
    for spec in "$@"; do IFS="|" read -r n p f <<<"$spec"; last_change[$n]=$(now); done
    while true; do
      for spec in "$@"; do
        IFS="|" read -r n p f <<<"$spec"
        tailline=$(grep -v '^[[:space:]]*$' "$f" 2>/dev/null | tail -1)
        if [[ -n "$BASE_HEAD" ]] && [[ "$(git -C "${f:h:h:h}" rev-parse HEAD 2>/dev/null)" == "$BASE_HEAD"* ]]; then tailline=""; fi
        case "$tailline" in DONE*|BLOCKED*) echo "$n: Handoff ends: $tailline"; exit 0 ;; esac
        astatus=$(herdr agent list 2>/dev/null | tr '{' '\n' | grep "\"name\":\"$n\"" | grep -o '"agent_status":"[^"]*"' | head -1)
        if [[ "$astatus" == *blocked* ]]; then
          [[ -z "${blocked_since[$n]}" ]] && blocked_since[$n]=$(now)
          (( $(now) - ${blocked_since[$n]} >= ${BLOCK_SEC:-10} )) && { echo "$n: blocked (approval waiting in pane $p)"; exit 0; }
        else blocked_since[$n]=""; fi
        text=$(herdr pane read "$p" 2>/dev/null | md5)
        if [[ "$text" != "${last_text[$n]}" ]]; then last_text[$n]=$text; last_change[$n]=$(now); fi
        (( $(now) - ${last_change[$n]} >= 1200 )) && { echo "$n: pane $p unchanged for 20 minutes"; exit 0; }
      done
      sleep 15
    done

```bash
# watch-round.sh <agent-name> <pane> <worktree> <base-commit>
# Wakes the lead on a commit past BASE with the agent no longer working, an approval (blocked > 10 s),
# the agent gone, or 20 minutes of unchanged pane text. Waits out an approval already showing.
NAME=$1; PANE=$2; WT=$3; BASE=$4
status() { herdr agent list 2>/dev/null | ruby -rjson -e 'a=JSON.parse(STDIN.read)["result"]["agents"].find{|x|x["name"]==ARGV[0]}; puts(a ? a["agent_status"] : "gone")' "$NAME"; }
sleep 20; while [ "$(status)" = "blocked" ]; do sleep 10; done
last_hash=""; last_change=$(date +%s); blocked_since=0
while true; do
  st=$(status); n=$(git -C "$WT" rev-list --count "$BASE"..HEAD)
  if [ "$n" -gt 0 ] && [ "$st" != "working" ] && [ "$st" != "blocked" ]; then echo "COMMITTED ($n) status=$st"; exit 0; fi
  [ "$st" = "gone" ] && { echo "AGENT GONE"; exit 0; }
  if [ "$st" = "blocked" ]; then [ $blocked_since -eq 0 ] && blocked_since=$(date +%s)
    [ $(( $(date +%s) - blocked_since )) -ge 10 ] && { echo "BLOCKED ON APPROVAL"; herdr pane read "$PANE" --lines 25 | tail -20; exit 0; }
  else blocked_since=0; fi
  h=$(herdr pane read "$PANE" --lines 40 2>/dev/null | md5); [ "$h" != "$last_hash" ] && { last_hash=$h; last_change=$(date +%s); }
  [ $(( $(date +%s) - last_change )) -ge 1200 ] && { echo "STALL 20 min"; exit 0; }
  sleep 15
done
```
